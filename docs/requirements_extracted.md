##### 5.2.5.7 RAG检索流程（编排器阶段1）
用户触发&quot;AI识别测试点&quot;
→ 编排器进入阶段1(RAG检索)
→ 将PRD/需求文本作为query向量化
→ pgvector检索知识库(按检索来源范围过滤)
→ 取相似度≥阈值的TopK条
→ 结果注入生成Prompt(按注入策略)
→ SSE推送&quot;检索到N条相关知识&quot;
→ 编排器进入阶段2(测试点分析)
降级方案：检索失败（超时/无结果/Embedding服务异常）时自动降级为&quot;仅PRD&quot;模式，标记阶段1状态为 degraded，不影响后五阶段。【deepseek】

vectorized_at TIMESTAMP,
fail_reason TEXT
);
CREATE INDEX idx_kb_doc_project ON knowledge_document(project_id);
CREATE INDEX idx_kb_doc_status ON knowledge_document(vector_status);
CREATE TABLE knowledge_chunk (
id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
document_id UUID NOT NULL REFERENCES knowledge_document(id) ON DELETE CASCADE,
project_id UUID NOT NULL REFERENCES project(id) ON DELETE CASCADE,
chunk_no INTEGER NOT NULL,
chunk_content TEXT NOT NULL,
embedding vector(1536),
source_type VARCHAR(20),
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
UNIQUE(document_id, chunk_no)
);
CREATE INDEX idx_kb_chunk_doc ON knowledge_chunk(document_id);
CREATE INDEX idx_kb_chunk_vec ON knowledge_chunk USING ivfflat (embedding vector_cosine_ops) WITH (lists=100);

id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

);
